class ManualSync { constructor() { this.records = []; this.loadLocal(); } // تولید کد اشتراک‌گذاری generateShareCode() { const data = { records: this.records, timestamp: Date.now(), version: '1.0' }; // فشرده‌سازی به Base64 const jsonStr = JSON.stringify(data); const base64 = btoa(unescape(encodeURIComponent(jsonStr))); return base64; } // وارد کردن از کد importFromCode(shareCode) { try { const jsonStr = decodeURIComponent(escape(atob(shareCode))); const data = JSON.parse(jsonStr); if (data.records && Array.isArray(data.records)) { this.records = data.records; this.saveLocal(); this.updateDisplay(); return true; } } catch (error) { console.error('خطا در وارد کردن کد:', error); } return false; } // نمایش رابط همگام‌سازی showSyncInterface() { const shareCode = this.generateShareCode(); document.body.innerHTML += `

🔄 همگام‌سازی بین دستگاه‌ها

📤 برای ارسال:

📥 برای دریافت:

`; } loadLocal() { this.records = JSON.parse(localStorage.getItem('ups_records') || '[]'); } saveLocal() { localStorage.setItem('ups_records', JSON.stringify(this.records)); } updateDisplay() { // کد نمایش رکوردها const recordsHTML = this.records.map(record => `
${record.branchName} (${record.branchCode})
UPS: ${record.upsModel}
${record.timestamp}
`).join(''); document.getElementById('recordsList').innerHTML = recordsHTML || '

هیچ رکوردی ثبت نشده است

'; document.getElementById('recordCount').textContent = this.records.length; } addRecord(record) { this.records.unshift({ id: Date.now(), ...record, timestamp: new Date().toLocaleString('fa-IR') }); this.saveLocal(); this.updateDisplay(); } } // ایجاد نمونه const syncManager = new ManualSync(); // توابع全局 function copyShareCode() { const textarea = document.getElementById('shareCode'); textarea.select(); document.execCommand('copy'); alert('✅ کد کپی شد! این کد را برای دستگاه دیگر بفرستید.'); } function importShareCode() { const importCode = document.getElementById('importCode').value.trim(); if (!importCode) { alert('❌ لطفاً کد را وارد کنید'); return; } if (syncManager.importFromCode(importCode)) { alert('✅ داده‌ها با موفقیت وارد شدند!'); closeSyncModal(); } else { alert('❌ کد نامعتبر است!'); } } function closeSyncModal() { document.getElementById('syncModal').remove(); } function showSyncMenu() { syncManager.showSyncInterface(); } function addRecord(record) { syncManager.addRecord(record); } function updateDisplay() { syncManager.updateDisplay(); }
